{{/* Returns a page's featured photo — the one its homepage card shows — as a dict with the proxy URLs, or nothing if the page has no album. url the API's base URL for this photo; append "/thumb.jpg", "/full.jpg" (or later sizes) to address one photo the album entry (photoGuid, caption, width, height, thumbWidth, …) .Params.featured indexes the album, oldest photo first. Call it through partialCached with the page as the variant, so a page that needs it twice (card and ) resolves it — and warns about it — once. */}} {{ $result := dict }} {{ with .Params.album }} {{ $album := . }} {{ $api := site.Params.icloud_api }} {{ $photos := partial "album-photos.html" (dict "api" $api "album" $album "where" $.RelPermalink) }} {{ if $photos }} {{/* `with` rather than a default: featured: 0 is a real value and would be treated as unset by anything truthiness-based. */}} {{ $i := 0 }} {{ with $.Params.featured }}{{ $i = int . }}{{ end }} {{ if ge $i (len $photos) }} {{ warnf "featured-photo: %s asks for featured photo %d but album %q has only %d — using the last one" $.RelPermalink $i $album (len $photos) }} {{ $i = sub (len $photos) 1 }} {{ end }} {{ $photo := index $photos $i }} {{ $result = dict "photo" $photo "url" (printf "%s/img/%s/%s" $api $album $photo.photoGuid) }} {{ end }} {{ end }} {{ return $result }}
