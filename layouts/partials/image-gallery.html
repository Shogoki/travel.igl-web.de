{{/* Renders a post's iCloud shared-album gallery as static markup. The album is fetched here, during the build, rather than by the browser on every page load. That is possible because the images come from the API's image proxy, whose URLs are addressed by photo GUID and never expire — unlike iCloud's own asset URLs, which are signed and die after about three hours, and so can never be written into a page. A failed fetch fails the build on purpose. The alternative is a deploy that quietly ships posts with no photos, which is far harder to notice; a failed build just needs running again once the API is back. */}} {{ $album := .album | default "" | string }} {{ if $album }} {{ $api := .site.Params.icloud_api }} {{ $photos := slice }} {{ $try := try (resources.GetRemote (printf "%s/album/%s" $api $album)) }} {{ with $try.Err }} {{ errorf "image-gallery: album %q could not be fetched: %s" $album . }} {{ else with $try.Value }} {{ $photos = . | transform.Unmarshal }} {{ else }} {{ errorf "image-gallery: album %q returned nothing (page: %s)" $album $.context.RelPermalink }} {{ end }} {{ if $photos }} {{/* The proxy is addressed by photo GUID, and the tiles are sized from the dimensions, so an API that predates either produces URLs reading "%!s()" and a silently broken gallery. Refuse to build instead. */}} {{ $sample := index $photos 0 }} {{ if or (not $sample.photoGuid) (not $sample.thumbWidth) }} {{ errorf "image-gallery: album %q came back without photoGuid/dimensions — the deployed album API predates the image proxy. Deploy the API before this site." $album }} {{ end }}
{{ range $index, $photo := $photos }} {{/* The .jpg suffix is what makes these cacheable at the edge: a CDN's default cache level decides from the file extension, and an extensionless URL is treated as dynamic however long its max-age. */}} {{ $base := printf "%s/img/%s/%s" $api $album $photo.photoGuid }}
{{ end }}
{{ else }} {{ warnf "image-gallery: album %q contains no photos (page: %s)" $album .context.RelPermalink }} {{ end }} {{ end }}
