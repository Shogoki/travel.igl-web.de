{{/* World map of every country we've been to, drawn by static/js/country-map.js on the same self-hosted outline as the route map (no tile server, see route-map.html). Clicking a country opens its page in the "countries" taxonomy, i.e. the posts from there. The countries come from two places, so neither has to be complete: - the `countries` in post front matter, matched by German name against data/iso3166.yaml ("USA" -> us). A name it doesn't know is warned about; add it to that country's `de` list. - the stops in route.yaml (on the route-data branch), by their country_code, which adds a country we've been to but not written about yet. data/iso3166.yaml then gives each country's outline (`n`, the id in world-50m.json) and flag. Each country also carries its stops (n, lat, lon): where static/data/states-50m.json has states for it (USA, Kanada, Brasilien, Australien, Indonesien, ...), the JS fills in the states the stops lie in. Below the map, the same countries as a list: tiny ones like Costa Rica are hard to hit on a phone, and it works without JavaScript. */}} {{ $iso := hugo.Data.iso3166 }} {{ $codeOf := dict }} {{ range $cc, $c := $iso }} {{ range $c.de }}{{ $codeOf = merge $codeOf (dict (lower .) $cc) }}{{ end }} {{ end }} {{/* Per country code: the date we first got there (from a stop or a post, whichever is earlier), its stops, and its taxonomy term. */}} {{ $first := dict }} {{ $names := dict }} {{ $stops := dict }} {{ $terms := dict }} {{ $slugCode := dict }} {{ with hugo.Data.route }} {{ range .trips }} {{ range .stops }} {{ $cc := lower .country_code }} {{ if or (not (index $first $cc)) (lt .from (index $first $cc)) }} {{ $first = merge $first (dict $cc .from) }} {{ end }} {{ if not (index $names $cc) }}{{ $names = merge $names (dict $cc .country) }}{{ end }} {{ $stops = merge $stops (dict $cc ((index $stops $cc | default slice) | append (dict "n" .name "lat" .lat "lon" .lon))) }} {{ range .posts }}{{ $slugCode = merge $slugCode (dict (lower .) $cc) }}{{ end }} {{ end }} {{ end }} {{ end }} {{ range site.Taxonomies.countries }} {{ with .Page }} {{ $term := . }} {{ $cc := index $codeOf (lower .Title) }} {{ if not $cc }} {{/* A name the table doesn't know: try the stops its posts are on. */}} {{ range .Pages }} {{ with index $slugCode (lower .File.ContentBaseName) }}{{ $cc = . }}{{ break }}{{ end }} {{ end }} {{ end }} {{ if not $cc }} {{ warnf "country-map: country %q (front matter) isn't in data/iso3166.yaml; add it to that country's `de` names to put it on the map" .Title }} {{ continue }} {{ end }} {{ $terms = merge $terms (dict $cc $term) }} {{ with index .Pages.ByDate 0 }} {{ $d := .Date.Format "2006-01-02" }} {{ if or (not (index $first $cc)) (lt $d (index $first $cc)) }} {{ $first = merge $first (dict $cc $d) }} {{ end }} {{ end }} {{ end }} {{ end }} {{/* Oldest first, so a new country is appended and doesn't shift the colours of the others. */}} {{ $order := slice }} {{ range $cc, $d := $first }}{{ $order = $order | append (dict "cc" $cc "d" $d) }}{{ end }} {{ $countries := slice }} {{ range sort $order "d" }} {{ $cc := .cc }} {{ $c := index $iso $cc }} {{ if not $c }} {{ warnf "country-map: unknown country code %q in route.yaml;
